import json
import ui, api, helpers as f, media
from snapshot_all import snapshot
L=ui.L;name='Empty-remove'
assert not (L/(name+'-result.json')).exists()
f.trash_page();ui.fill(0,name);ui.device('shell','input','keyevent','4')
f.until(lambda n:'共 1 个图包' in f.labels(n),'empty target')
ui.choose('选择回收内容 '+name)
before=media.save(name+'-before');snapshot('normal26',name+'-before')
seq=max(json.loads(x)['seq'] for x in (L/'requests.jsonl').read_text().splitlines())
ui.choose('永久删除');f.until(lambda n:any('我确认永久删除' in x for x in f.labels(n)),'image review')
ui.snap(name+'-review');ui.choose('我确认永久删除本批清单中的图片，无法恢复。');ui.choose('确认永久删除')
f.until(lambda n:'已确认删除 2/2 张' in f.labels(n),'two images deleted')
ui.snap(name+'-completed');after=media.save(name+'-after');media.compare_deleted(before,after,[name+'/01.png',name+'/02.png'])
assert not (L/'media'/name).exists()
trace=[json.loads(x) for x in (L/'requests.jsonl').read_text().splitlines()]
posts=[x for x in trace if x['seq']>seq and x['path']=='/api/v1/pack-trash/permanent' and x['method']=='POST'];assert len(posts)==1
task=api.call('GET','/api/v1/pack-file-requests/'+posts[0]['command']['request_id']);assert task['status']=='completed' and task['result']['pages_deleted']==2
snapshot('normal26',name+'-after')
result={'name':name,'task':task,'only_two_confirmed_images_removed':True,'empty_root_and_child_removed':True,'other_files_byte_identical':True,'native_permanent_posts':1}
(L/(name+'-result.json')).write_text(json.dumps(result,indent=2));ui.choose('关闭');ui.choose('返回设置');print(result,flush=True)
